3  index.ts 파일 생성 (위치: expressjs_backend/src/ 내부)

import express from 'express';
import cors from 'cors';
import dotenv from 'dotenv';
import { connectDB } from './config/db.js'; // 꼭 뒤에 .js를 붙여주세요! (TS의 정석 규칙입니다)

dotenv.config();

const app = express();
const PORT = process.env.PORT || 5000;

app.use(cors());
app.use(express.json());

// 1. 데이터베이스 커넥션
connectDB();

// 2. 기본 API 엔드포인트
app.get('/api/health', (req: any, res: any) => {
  res.json({ 
    status: 'UP', 
    message: 'Express.js 백엔드가 정상동작 중입니다.' 
  });
});

try {
  app.listen(PORT, () => {
    console.log(`🚀 [SUCCESS] Express 서버가 포트 ${PORT}에서 상시 대기 모드로 돌입했습니다.`);
  });
} catch (error) {
  console.error('⚠️ 서버 구동 중 치명적 오류 발생:', error);
}

---테이블 생성

-- 1. robotics 업무용 스키마 생성
CREATE SCHEMA IF NOT EXISTS robotics;

-- 2. 기존 테이블이 존재할 경우 안전하게 삭제
DROP TABLE IF EXISTS robotics.robot_logs;

-- 3. robotics 스키마 내에 테이블 생성
CREATE TABLE robotics.robot_logs (
    id SERIAL PRIMARY KEY,
    robot_name VARCHAR(50) NOT NULL,
    status VARCHAR(30) NOT NULL,
    command VARCHAR(255),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 4. [보안 강화] Row Level Security(RLS) 활성화
-- 이 설정을 켜면 Supabase의 기본 anon(익명) 키나 일반 인증 키로는 이 테이블을 읽거나 쓸 수 없습니다.
ALTER TABLE robotics.robot_logs ENABLE ROW LEVEL SECURITY;

-- 5. [보안 강화] 오직 서버 백엔드(Service Role / Postgres 마스터 계정)만 접근 가능하도록 허용
-- 서비스 외부에서 anon 키로 해킹 시도를 해도 무조건 차단됩니다.
CREATE POLICY "Allow full access to postgres admin only" 
ON robotics.robot_logs
TO postgres
USING (true)
WITH CHECK (true);
