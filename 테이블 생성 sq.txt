CREATE TABLE TB_CURRICULUM (
    CUR_SEQ      SERIAL PRIMARY KEY,
    ROOM         VARCHAR(50),
    TERM         INTEGER,
    MAN_COUNT    INTEGER,
    START_DATE   DATE NOT NULL,
    END_DATE     DATE NOT NULL
);

-- 2. 과목 테이블
CREATE TABLE TB_SUBJECT (
    SUB_SEQ      SERIAL PRIMARY KEY,
    CUR_SEQ      INTEGER NOT NULL,
    SUB_NAME     VARCHAR(100) NOT NULL,
    START_DATE   DATE,
    END_DATE     DATE,
    CONSTRAINT FK_SUBJECT_CURRICULUM FOREIGN KEY (CUR_SEQ) 
        REFERENCES TB_CURRICULUM (CUR_SEQ) ON DELETE CASCADE
);

-- 3. 학생 테이블 (기존 4번)
CREATE TABLE TB_STUDENT (
    STU_SEQ      SERIAL PRIMARY KEY,
    CUR_SEQ      INTEGER NOT NULL,
    USER_ID      VARCHAR(50) UNIQUE NOT NULL,
    USER_NAME    VARCHAR(50) NOT NULL,
    BIRTH        DATE,
    GENDER       CHAR(1),
    TEL          VARCHAR(20),
    EDU          VARCHAR(50),
    MAJOR        VARCHAR(50),
    GRAD_TYPE    VARCHAR(20),
    CAREER       TEXT,
    DROPOUT_INFO TEXT,
    CONSTRAINT FK_STUDENT_CURRICULUM FOREIGN KEY (CUR_SEQ) 
        REFERENCES TB_CURRICULUM (CUR_SEQ)
);

-- 4. 테스트(시험지) 테이블 (기존 1번)
CREATE TABLE TB_TEST (
    TEST_SEQ      SERIAL PRIMARY KEY,
    SUB_SEQ       INTEGER NOT NULL,
    TEST_NAME     VARCHAR(200) NOT NULL,
    DURATION      INTEGER,
    START_TIME    TIMESTAMP,
    END_TIME      TIMESTAMP,
    CONSTRAINT FK_TEST_SUBJECT FOREIGN KEY (SUB_SEQ) REFERENCES TB_SUBJECT (SUB_SEQ) ON DELETE CASCADE
);

-- 5. 문제 테이블
CREATE TABLE TB_QUESTION (
    QUES_SEQ      SERIAL PRIMARY KEY,
    TEST_SEQ      INTEGER NOT NULL,
    QUES_NUM      INTEGER NOT NULL,
    CONTENT       TEXT NOT NULL,
    QUES_TYPE     VARCHAR(20),
    ANSWER        TEXT NOT NULL,
    POINT         INTEGER DEFAULT 5,
    CONSTRAINT FK_QUES_TEST FOREIGN KEY (TEST_SEQ) REFERENCES TB_TEST (TEST_SEQ) ON DELETE CASCADE
);

-- 6. 보기 테이블
CREATE TABLE TB_CHOICE (
    CHOICE_SEQ    SERIAL PRIMARY KEY,
    QUES_SEQ      INTEGER NOT NULL,
    CHOICE_NUM    INTEGER NOT NULL,
    CONTENT       TEXT NOT NULL,
    CONSTRAINT FK_CHOICE_QUES FOREIGN KEY (QUES_SEQ) REFERENCES TB_QUESTION (QUES_SEQ) ON DELETE CASCADE
);

-- 7. 학생 답안 제출 테이블
CREATE TABLE TB_STUDENT_ANSWER (
    ANS_SEQ       SERIAL PRIMARY KEY,
    STU_SEQ       INTEGER NOT NULL,
    QUES_SEQ      INTEGER NOT NULL,
    SUBMITTED_ANS TEXT,
    IS_CORRECT    BOOLEAN,
    CONSTRAINT FK_ANS_STUDENT FOREIGN KEY (STU_SEQ) REFERENCES TB_STUDENT (STU_SEQ),
    CONSTRAINT FK_ANS_QUES FOREIGN KEY (QUES_SEQ) REFERENCES TB_QUESTION (QUES_SEQ)
);

-- 8. 최종 성적 테이블
CREATE TABLE TB_TEST_RESULT (
    RESULT_SEQ    SERIAL PRIMARY KEY,
    STU_SEQ       INTEGER NOT NULL,
    TEST_SEQ      INTEGER NOT NULL,
    TOTAL_SCORE   INTEGER DEFAULT 0,
    STATUS        VARCHAR(20),
    REG_DATE      TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT FK_RESULT_STUDENT FOREIGN KEY (STU_SEQ) REFERENCES TB_STUDENT (STU_SEQ),
    CONSTRAINT FK_RESULT_TEST FOREIGN KEY (TEST_SEQ) REFERENCES TB_TEST (TEST_SEQ)
);

  start_time TIMESTAMP WITHOUT TIME ZONE NULL,
